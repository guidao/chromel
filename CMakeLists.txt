cmake_minimum_required(VERSION 3.21)
project(emacs_web LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "This native backend currently supports macOS only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CEF_ROOT "" CACHE PATH "Extracted CEF minimal binary distribution")
set(EMACS_INCLUDE_DIR "" CACHE PATH "Directory containing emacs-module.h")
if(NOT EXISTS "${CEF_ROOT}/include/cef_app.h")
  message(FATAL_ERROR "Set CEF_ROOT or run scripts/build.sh")
endif()
if(NOT EXISTS "${EMACS_INCLUDE_DIR}/emacs-module.h")
  message(FATAL_ERROR "Set EMACS_INCLUDE_DIR to the installed Emacs include directory")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)

add_library(web_core STATIC native/web_core.mm)
target_include_directories(web_core PUBLIC "${CEF_ROOT}" native)
target_compile_options(web_core PRIVATE -fobjc-arc -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
target_link_libraries(web_core PUBLIC libcef_dll_wrapper "-framework Cocoa")

add_library(web-module MODULE native/module.mm)
target_include_directories(web-module PRIVATE "${EMACS_INCLUDE_DIR}")
target_link_libraries(web-module PRIVATE web_core)
set_target_properties(web-module PROPERTIES PREFIX "" SUFFIX ".dylib")

add_executable(web-helper native/helper.cc)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/scroll.js")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native/scroll.js" WEB_SCROLL_SOURCE)
configure_file(native/scripts.h.in generated/web_scripts.h @ONLY)
target_include_directories(web-helper PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(web-helper PRIVATE libcef_dll_wrapper "-framework Cocoa")

add_executable(web-smoke native/smoke.mm)
target_compile_options(web-smoke PRIVATE -fobjc-arc)
target_link_libraries(web-smoke PRIVATE web_core)

# Explicit opt-in regression probe; never packaged in the runtime or module.
add_library(web-pump-probe MODULE EXCLUDE_FROM_ALL test/pump-probe.mm)
target_include_directories(web-pump-probe PRIVATE "${EMACS_INCLUDE_DIR}")
target_link_libraries(web-pump-probe PRIVATE "-framework Cocoa")
set_target_properties(web-pump-probe PROPERTIES PREFIX "" SUFFIX ".dylib")
